const LS_KEY = "meridian-cities-v1";
const localTz = Intl.DateTimeFormat().resolvedOptions().timeZone || "UTC";
const state = { cities: [], offsetH: 0, drag: null };

/* --------------------------- format helpers -------------------------- */
const fmtCache = new Map();
function fmtFor(tz){
  if(!fmtCache.has(tz)){
    fmtCache.set(tz,{
      time:new Intl.DateTimeFormat("en-GB",{timeZone:tz,hourCycle:"h23",hour:"2-digit",minute:"2-digit",second:"2-digit"}),
      hm:  new Intl.DateTimeFormat("en-GB",{timeZone:tz,hourCycle:"h23",hour:"2-digit",minute:"2-digit"}),
      date:new Intl.DateTimeFormat("en-GB",{timeZone:tz,weekday:"short",day:"numeric",month:"short"}),
      off: new Intl.DateTimeFormat("en-US",{timeZone:tz,timeZoneName:"shortOffset"}),
      ymd: new Intl.DateTimeFormat("en-CA",{timeZone:tz,year:"numeric",month:"2-digit",day:"2-digit"})
    });
  }
  return fmtCache.get(tz);
}
function parts(f,d){const o={};for(const p of f.formatToParts(d))o[p.type]=p.value;return o;}
function offsetLabel(tz,d){
  const p=parts(fmtFor(tz).off,d).timeZoneName||"GMT";
  return p.replace("GMT","UTC").replace(":00","");
}
function tzCityName(tz){
  return tz.split("/").pop().replace(/_/g," ");
}

/* --------------------------- card construction ----------------------- */
function buildTimeline(){
  const t=document.createElement("div");
  t.className="timeline";t.tabIndex=0;t.setAttribute("role","slider");
  t.setAttribute("aria-label","Drag to shift time for all cities");
  t.setAttribute("aria-valuemin","0");t.setAttribute("aria-valuemax","24");
  let cells="";
  for(let h=0;h<24;h++){
    const night=(h<6||h>=19)?" night":"";
    cells+=`<div class="hcell${night}" title="${String(h).padStart(2,"0")}:00" aria-hidden="true"></div>`;
  }
  let ticks="";
  for(let h=0;h<24;h++) ticks+=`<span class="tick">${String(h).padStart(2,"0")}</span>`;
  t.innerHTML=`
    <div class="hours">${cells}</div>
    <div class="marker"><span class="m-label"></span></div>
    <div class="ticks">${ticks}</div>`;
  /* drag wiring */
  t.addEventListener("pointerdown",e=>{
    t.setPointerCapture(e.pointerId);
    state.drag={el:t,startX:e.clientX,rect:t.getBoundingClientRect(),startOff:state.offsetH};
  });
  t.addEventListener("pointermove",e=>{
    if(!state.drag||state.drag.el!==t)return;
    const dx=e.clientX-state.drag.startX;
    state.offsetH=state.drag.startOff+dx*24/state.drag.rect.width;
    updateClocks();
  });
  const end=()=>{state.drag=null;};
  t.addEventListener("pointerup",end);
  t.addEventListener("pointercancel",end);
  /* keyboard support: arrows shift 30 min */
  t.addEventListener("keydown",e=>{
    if(e.key==="ArrowRight"){state.offsetH+=0.5;e.preventDefault();updateClocks();}
    if(e.key==="ArrowLeft"){state.offsetH-=0.5;e.preventDefault();updateClocks();}
  });
  return t;
}

function renderGrid(){
  grid.innerHTML="";
  state.cardRefs=[];
  state.cardRefs.push({...makeCard(null,true),tz:localTz,isRef:true});
  for(const c of state.cities){
    state.cardRefs.push({...makeCard(c,false),tz:c.tz,isRef:false});
  }
  updateClocks();
}

/* ------------------------------ updating ----------------------------- */
function updateClocks(){
  const now=new Date(Date.now()+state.offsetH*3600e3);
  const refYmd=parts(fmtFor(localTz).ymd,now).year+"-"+parts(fmtFor(localTz).ymd,now).month+"-"+parts(fmtFor(localTz).ymd,now).day;

  for(const r of state.cardRefs){
    const f=fmtFor(r.tz);
    const tp=parts(f.time,now);
    r.clock.textContent=`${tp.hour}:${tp.minute}:${tp.second}`;
    r.date.textContent=f.date.format(now);
    r.off.textContent=offsetLabel(r.tz,now);
    /* day difference vs reference */
    const ymd=parts(f.ymd,now);
    const key=`${ymd.year}-${ymd.month}-${ymd.day}`;
    if(!r.isRef&&key!==refYmd){
      const a=new Date(key+"T12:00:00Z"),b=new Date(refYmd+"T12:00:00Z");
      const d=Math.round((a-b)/864e5);
      r.diff.hidden=false;
      r.diff.textContent=d>0?`+${d}d`:`${d}d`;
      r.diff.setAttribute("aria-label",d>0?`One day ahead of reference`:"One day behind reference");
    }else{r.diff.hidden=true;}
    /* marker position */
    const pos=((+tp.hour)+(+tp.minute)/60+(+tp.second)/3600)/24*100;
    r.marker.style.left=pos+"%";
    r.mlabel.textContent=f.hm.format(now);
    r.timeline.setAttribute("aria-valuenow",(pos/100*24).toFixed(1));
  }
  /* shift chip */
  if(Math.abs(state.offsetH)<0.001){
    shiftChip.classList.remove("on");
  }else{
    shiftChip.classList.add("on");
    const h=state.offsetH;
    shiftText.textContent=`${h>0?"+":"−"}${Math.abs(h).toFixed(1)}h from live`;
  }
}

/* --------------------------- search / add ---------------------------- */
function renderResults(){
  const q=searchInput.value.trim().toLowerCase();
  clearBtn.style.display=q?"inline-block":"none";
  const added=new Set(state.cities.map(c=>c.tz));
  if(localTz)added.add(localTz);
  const matches=CITIES.filter(c=>
    !added.has(c.tz)&&
    (c.name.toLowerCase().includes(q)||c.country.toLowerCase().includes(q)||c.tz.toLowerCase().includes(q))
  );
  if(!q){resultsBox.classList.remove("open");resultsBox.innerHTML="";return;}
  resultsBox.classList.add("open");
  if(!matches.length){
    resultsBox.innerHTML=`<div class="empty-note">No matching city${q?` for “${searchInput.value.trim()}”`:""}.</div>`;
    return;
  }
  resultsBox.innerHTML="";
  matches.slice(0,12).forEach(c=>{
    const b=document.createElement("button");
    b.className="result";b.setAttribute("role","option");
    b.innerHTML=`<div><div class="rname">${c.name}</div><div class="rtz">${c.tz}</div></div><span class="add-tag">+ Add</span>`;
    b.addEventListener("click",()=>addCity(c));
    resultsBox.appendChild(b);
  });
}
function addCity(c){
  state.cities.push(c);
  save();renderGrid();renderResults();
  searchInput.value="";renderResults();
}
searchInput.addEventListener("input",renderResults);
searchInput.addEventListener("keydown",e=>{
  if(e.key==="Enter"){
    const first=resultsBox.querySelector(".result");
    if(first)first.click();
  }
  if(e.key==="Escape"){searchInput.value="";renderResults();searchInput.blur();}
});
clearBtn.addEventListener("click",()=>{searchInput.value="";renderResults();searchInput.focus();});
document.addEventListener("click",e=>{
  if(!e.target.closest(".add-box"))resultsBox.classList.remove("open");
});

/* ------------------------------ live --------------------------------- */

  state.offsetH=0;updateClocks();
  liveBtn.animate(
    [{transform:"scale(1)"},{transform:"scale(.9)"},{transform:"scale(1)"}],
    {duration:220,easing:"ease-out"});
});

/* ---------------------------- persistence ---------------------------- */
function save(){
  try{localStorage.setItem(LS_KEY,JSON.stringify(state.cities.map(c=>c.tz)));}catch(e){}
}
function load(){
  try{
    const arr=JSON.parse(localStorage.getItem(LS_KEY)||"[]");
    arr.forEach(tz=>{
      const c=CITIES.find(x=>x.tz===tz);
      if(c)state.cities.push(c);
    });
  }catch(e){}
  if(!state.cities.length){
    ["Europe/London","America/New_York","Asia/Tokyo"].forEach(tz=>{
      if(tz!==localTz){
        const c=CITIES.find(x=>x.tz===tz);
        if(c)state.cities.push(c);
      }
    });
  }
}
